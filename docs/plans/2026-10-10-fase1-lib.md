**Goal:** Ship the `glassllm` Python library: BPE tokenizer, NumPy autograd, a traceable Transformer with KV cache, training, a CLI and pretrained PT/EN weights, ready for PyPI.

**Architecture:** One package under `src/glassllm/`, NumPy as the only runtime dependency. **Tech Stack:** Python ≥ 3.10, NumPy, uv + hatchling, pytest, PyTorch CPU (tests only), GitHub Actions.

```python
from glassllm.autograd import (Tensor, cross_entropy, embedding, exp, log, masked_fill, no_grad,
                               repeat_kv, rope, silu, softmax, transpose)

RNG = np.random.default_rng(0)


def numeric_grad(f, x, eps=1e-6):
    """Central finite difference of the scalar f() with respect to every entry of x."""
    grad = np.zeros_like(x)
    it = np.nditer(x, flags=["multi_index"])
    for _ in it:
        i = it.multi_index
        old = x[i]
        x[i] = old + eps
        plus = f()
        x[i] = old - eps
        minus = f()
        x[i] = old
        grad[i] = (plus - minus) / (2 * eps)
    return grad


def check(op, *shapes, positive=False):
    """Compare autograd with finite differences for loss = sum(op(*inputs) * w)."""
    arrays = [RNG.normal(size=s) for s in shapes]
    if positive:
        arrays = [np.abs(a) + 0.5 for a in arrays]
    tensors = [Tensor(a, requires_grad=True) for a in arrays]
    out = op(*tensors)
    w = RNG.normal(size=out.shape)
    (out * w).sum().backward()
    for a, t in zip(arrays, tensors):
        expected = numeric_grad(lambda: float((op(*[Tensor(b) for b in arrays]).data * w).sum()), a)
```

`src/glassllm/autograd.py`:

```python
"""A tiny reverse-mode autograd engine on top of NumPy.

Each Tensor remembers the tensors it came from and a function that pushes its
gradient back to them. backward() walks that graph in reverse topological
order. It is the same idea as micrograd, with arrays instead of scalars.
"""
from __future__ import annotations

import contextlib

import numpy as np

_grad_enabled = True


@contextlib.contextmanager
def no_grad():
    """Run without recording the graph (inference)."""
    global _grad_enabled
    previous = _grad_enabled
    _grad_enabled = False
    try:
        yield
    finally:
        _grad_enabled = previous


def _unbroadcast(grad, shape):
    # Broadcasting copied a value along some axes; its gradient is the sum over them.
    while grad.ndim > len(shape):
        grad = grad.sum(axis=0)
    for axis, size in enumerate(shape):
        if size == 1 and grad.shape[axis] != 1:
            grad = grad.sum(axis=axis, keepdims=True)
    return grad


class Tensor:
    def __init__(self, data, parents=(), backward=None, requires_grad=False):
        self.data = np.asarray(data)
        self.grad = None
        self.requires_grad = requires_grad or any(p.requires_grad for p in parents)
        track = _grad_enabled and self.requires_grad
        self._parents = tuple(parents) if track else ()
        self._backward = backward if track else None

    @property
    def shape(self):
        return self.data.shape

    @property
    def ndim(self):
        return self.data.ndim

    def __repr__(self):
        return f"Tensor(shape={self.shape}, requires_grad={self.requires_grad})"

    def _accumulate(self, grad):
        if not self.requires_grad:
            return
        grad = _unbroadcast(grad, self.shape)
        self.grad = grad if self.grad is None else self.grad + grad

    def backward(self, grad=None):
        if grad is None:
            grad = np.ones_like(self.data)
        order, seen, stack = [], set(), [(self, False)]
        while stack:
            node, expanded = stack.pop()
            if expanded:
                order.append(node)
                continue
            if id(node) in seen:
                continue
            seen.add(id(node))
            stack.append((node, True))
            stack.extend((parent, False) for parent in node._parents)
        self.grad = np.asarray(grad)
        for node in reversed(order):
            if node._backward is not None and node.grad is not None:
                node._backward(node.grad)

    # operators
    def __add__(self, other):
        return add(self, other)

    __radd__ = __add__

    def __sub__(self, other):
        return add(self, neg(_t(other)))

    def __rsub__(self, other):
        return add(_t(other), neg(self))

    def __mul__(self, other):
        return mul(self, other)

    __rmul__ = __mul__

    def __truediv__(self, other):
        return div(self, other)

    def __neg__(self):
        return neg(self)

    def __pow__(self, exponent):
        return power(self, exponent)

    def __matmul__(self, other):
        return matmul(self, other)

    def sum(self, axis=None, keepdims=False):
        return tsum(self, axis, keepdims)

    def mean(self, axis=None, keepdims=False):
        return mean(self, axis, keepdims)

    def reshape(self, *shape):
        return reshape(self, shape)

    def transpose(self, *axes):
        return transpose(self, axes)


def _t(x):
    return x if isinstance(x, Tensor) else Tensor(x)


def add(a, b):
    a, b = _t(a), _t(b)

    def back(g):
        a._accumulate(g)
        b._accumulate(g)

    return Tensor(a.data + b.data, (a, b), back)


def neg(a):
    def back(g):
        a._accumulate(-g)

    return Tensor(-a.data, (a,), back)


def mul(a, b):
    a, b = _t(a), _t(b)

    def back(g):
        a._accumulate(g * b.data)
        b._accumulate(g * a.data)

    return Tensor(a.data * b.data, (a, b), back)


def div(a, b):
    a, b = _t(a), _t(b)

    def back(g):
        a._accumulate(g / b.data)
        b._accumulate(-g * a.data / b.data**2)

    return Tensor(a.data / b.data, (a, b), back)


def power(a, exponent):
    def back(g):
        a._accumulate(g * exponent * a.data ** (exponent - 1))

    return Tensor(a.data**exponent, (a,), back)


def matmul(a, b):
    a, b = _t(a), _t(b)

    def back(g):
        a._accumulate(g @ np.swapaxes(b.data, -1, -2))
        b._accumulate(np.swapaxes(a.data, -1, -2) @ g)

    return Tensor(a.data @ b.data, (a, b), back)


def exp(a):
    y = np.exp(a.data)

    def back(g):
        a._accumulate(g * y)

    return Tensor(y, (a,), back)


def log(a):
    def back(g):
        a._accumulate(g / a.data)

    return Tensor(np.log(a.data), (a,), back)


def tsum(a, axis=None, keepdims=False):
    def back(g):
        if axis is not None and not keepdims:
            g = np.expand_dims(g, axis)
        a._accumulate(np.broadcast_to(g, a.shape))

    return Tensor(a.data.sum(axis=axis, keepdims=keepdims), (a,), back)


def mean(a, axis=None, keepdims=False):
    count = a.data.size if axis is None else np.prod([a.shape[i] for i in np.atleast_1d(axis)])
    return tsum(a, axis, keepdims) / float(count)


def reshape(a, shape):
    def back(g):
        a._accumulate(g.reshape(a.shape))

    return Tensor(a.data.reshape(shape), (a,), back)


def transpose(a, axes):
    def back(g):
        a._accumulate(g.transpose(np.argsort(axes)))

    return Tensor(a.data.transpose(axes), (a,), back)


def embedding(weight, ids):
    """Row lookup E[ids]. Same as one_hot(ids) @ E, without building the one-hot."""
    ids = np.asarray(ids)

    def back(g):
        grad = np.zeros_like(weight.data)
        np.add.at(grad, ids, g)  # a token that appears twice gets both gradients
        weight._accumulate(grad)

    return Tensor(weight.data[ids], (weight,), back)


def masked_fill(a, mask, value):
    def back(g):
        a._accumulate(np.where(mask, 0.0, g))

    return Tensor(np.where(mask, value, a.data), (a,), back)


def softmax_np(x, axis=-1):
    # Subtracting the max does not change the result and avoids overflow in exp.
    z = x - x.max(axis=axis, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=axis, keepdims=True)


def softmax(a, axis=-1):
    y = softmax_np(a.data, axis)

    def back(g):
        # dp_i/ds_j = p_i (delta_ij - p_j)  =>  grad_s = p * (g - sum(g * p))
        a._accumulate(y * (g - (g * y).sum(axis=axis, keepdims=True)))

    return Tensor(y, (a,), back)


def silu(a):
    s = 0.5 * (1.0 + np.tanh(0.5 * a.data))  # sigmoid, written so it never overflows

    def back(g):
        a._accumulate(g * s * (1.0 + a.data * (1.0 - s)))

    return Tensor(a.data * s, (a,), back)


def cross_entropy_grad(logits, targets):
    """Gradient of the mean cross-entropy with respect to the logits: (p - y) / N."""
    p = softmax_np(logits, axis=-1)
    p[np.arange(len(targets)), targets] -= 1.0  # @step loss.grad
    return p / len(targets)


def cross_entropy(logits, targets):
    """Mean of -log softmax(logits)[target]. logits: (N, V), targets: (N,)."""
    targets = np.asarray(targets)
    z = logits.data - logits.data.max(axis=-1, keepdims=True)
    log_p = z - np.log(np.exp(z).sum(axis=-1, keepdims=True))
    loss = -log_p[np.arange(len(targets)), targets].mean()

    def back(g):
        logits._accumulate(g * cross_entropy_grad(logits.data, targets))

    return Tensor(loss, (logits,), back)


def _rotate(x, cos, sin):
    x_even, x_odd = x[..., 0::2], x[..., 1::2]
    out = np.empty_like(x)
    out[..., 0::2] = x_even * cos - x_odd * sin
    out[..., 1::2] = x_even * sin + x_odd * cos
    return out


def rope(x, cos, sin):
    """Rotate each pair (x_2i, x_2i+1) by the angle m * theta_i."""

    def back(g):
        # The transpose of a rotation is the rotation by the opposite angle.
        x._accumulate(_rotate(g, cos, -sin))

    return Tensor(_rotate(x.data, cos, sin), (x,), back)


def repeat_kv(x, repeats):
    """GQA: (B, H_kv, n, d_k) -> (B, H_kv * repeats, n, d_k); each K/V head serves `repeats` query heads."""
    if repeats == 1:
        return x
    b, h_kv, n, d_k = x.shape

    def back(g):
        x._accumulate(g.reshape(b, h_kv, repeats, n, d_k).sum(axis=2))

    return Tensor(np.repeat(x.data, repeats, axis=1), (x,), back)
```

`src/glassllm/tokenizer.py`:

```python
"""Byte-pair encoding over characters, in the style of SentencePiece (used by Llama).

Spaces become "▁" so they are ordinary symbols. Training starts from single
characters and keeps merging the most frequent adjacent pair until the
vocabulary reaches the target size. Merges never cross a word boundary.
"""
from __future__ import annotations

import json
import re
import unicodedata
from collections import Counter
from pathlib import Path

SPACE = "▁"
SPECIALS = ["<pad>", "<unk>", "<bos>", "<eos>"]
UNK_TEXT = "⁇"
_CHUNK = re.compile(r"\n|▁*[^▁\n]+|▁+")


def pre_tokenize(text: str) -> list[str]:
    """Normalize and split into chunks that merges are not allowed to cross."""
    text = unicodedata.normalize("NFC", text).replace(" ", SPACE)
    return _CHUNK.findall(text)


def _apply_merge(symbols: tuple[str, ...], pair: tuple[str, str]) -> tuple[str, ...]:
    out, i = [], 0
    while i < len(symbols):
        if i + 1 < len(symbols) and (symbols[i], symbols[i + 1]) == pair:
            out.append(symbols[i] + symbols[i + 1])
            i += 2
        else:
            out.append(symbols[i])
            i += 1
    return tuple(out)


class Tokenizer:
    def __init__(self, alphabet: list[str], merges: list[tuple[str, str]]):
        self.alphabet = list(alphabet)
        self.merges = [tuple(m) for m in merges]
        self.ranks = {pair: i for i, pair in enumerate(self.merges)}
        self.vocab = SPECIALS + self.alphabet + [a + b for a, b in self.merges]
        self.token_to_id = {tok: i for i, tok in enumerate(self.vocab)}
        self.unk_id = self.token_to_id["<unk>"]
        self._cache: dict[str, list[str]] = {}

    @property
    def vocab_size(self) -> int:
        return len(self.vocab)

    @classmethod
    def train(cls, text: str, vocab_size: int) -> "Tokenizer":
        words = Counter(tuple(chunk) for chunk in pre_tokenize(text))
        alphabet = sorted({ch for word in words for ch in word})
        n_merges = vocab_size - len(SPECIALS) - len(alphabet)
        if n_merges < 0:
            raise ValueError(f"vocab_size={vocab_size} is smaller than specials + alphabet ({len(SPECIALS) + len(alphabet)})")
        merges = []
        for _ in range(n_merges):
            pairs = Counter()
            for word, freq in words.items():
                for pair in zip(word, word[1:]):
                    pairs[pair] += freq
            if not pairs:
                break
            # Most frequent pair; ties broken alphabetically so training is deterministic.
            best = min(pairs, key=lambda p: (-pairs[p], p))
            merges.append(best)
            words = Counter({_apply_merge(w, best): f for w, f in words.items()})
        return cls(alphabet, merges)

    def _merge_chunk(self, chunk: str, steps: list | None = None) -> list[str]:
        if steps is None and chunk in self._cache:
            return self._cache[chunk]
        symbols = tuple(chunk)
        if steps is not None:
            steps.append(list(symbols))
        while len(symbols) > 1:
            # Apply the merge that was learned first (lowest rank), like during training.
            ranked = [(self.ranks[p], p) for p in zip(symbols, symbols[1:]) if p in self.ranks]
            if not ranked:
                break
            symbols = _apply_merge(symbols, min(ranked)[1])
            if steps is not None:
                steps.append(list(symbols))
        result = list(symbols)
        self._cache[chunk] = result
        return result

    def encode(self, text: str) -> list[int]:
        ids = []
        for chunk in pre_tokenize(text):
            for piece in self._merge_chunk(chunk):
                ids.append(self.token_to_id.get(piece, self.unk_id))
        return ids

    def encode_verbose(self, text: str) -> list[dict]:
        """For each chunk, every intermediate state from characters to final tokens."""
        out = []
        for chunk in pre_tokenize(text):
            steps: list[list[str]] = []
            self._merge_chunk(chunk, steps)
            out.append({"chunk": chunk, "steps": steps})
        return out

    def tokens(self, ids: list[int]) -> list[str]:
        return [self.vocab[i] for i in ids]

    def decode(self, ids: list[int]) -> str:
        pieces = []
        for i in ids:
            token = self.vocab[i]
            if token == "<unk>":
                pieces.append(UNK_TEXT)
            elif token not in SPECIALS:
                pieces.append(token)
        return "".join(pieces).replace(SPACE, " ")

    def save(self, path: str | Path) -> None:
        data = {"version": 1, "alphabet": self.alphabet, "merges": [list(m) for m in self.merges]}
        Path(path).write_text(json.dumps(data, ensure_ascii=False), encoding="utf-8")

    @classmethod
    def load(cls, path: str | Path) -> "Tokenizer":
        data = json.loads(Path(path).read_text(encoding="utf-8"))
        return cls(data["alphabet"], data["merges"])
```

```python
"""Debug mode: record the value of every marked step while the model runs."""
from __future__ import annotations

from dataclasses import dataclass, replace

import numpy as np

from .autograd import Tensor
from .steps import HEAD_STEPS, STEPS


@dataclass(frozen=True)
class StepRecord:
    id: str
    layer: int | None
    value: np.ndarray
    order: int
    head: int | None = None

    @property
    def shape(self) -> tuple[int, ...]:
        return self.value.shape

    @property
    def equation(self) -> str:
        return STEPS[self.id].plain

    @property
    def latex(self) -> str:
        return STEPS[self.id].latex


class Tracer:
    def __init__(self):
        self.records: list[StepRecord] = []

    def record(self, step_id: str, value, layer: int | None = None, batched: bool = True) -> None:
        if step_id not in STEPS:
            raise KeyError(f"unknown step id {step_id!r}; add it to glassllm/steps.py")
        data = value.data if isinstance(value, Tensor) else np.asarray(value)
        if batched:
            data = data[0]  # the tracer follows one sequence
        self.records.append(StepRecord(step_id, layer, np.array(data, copy=True), len(self.records)))

    def get(self, step_id: str, layer: int | None = None, head: int | None = None) -> StepRecord:
        for rec in self.records:
            if rec.id == step_id and (layer is None or rec.layer == layer):
                break
        else:
            raise KeyError(f"step {step_id!r} (layer={layer}) was not recorded")
        if head is None:
            return rec
        if step_id not in HEAD_STEPS:
            raise ValueError(f"step {step_id!r} has no head axis")
        return replace(rec, value=rec.value[head], head=head)

    def step_ids(self) -> list[str]:
        return list(dict.fromkeys(rec.id for rec in self.records))


def record(trace: Tracer | None, step_id: str, value, layer: int | None = None, batched: bool = True) -> None:
    """Record only when a tracer was given; costs nothing otherwise."""
    if trace is not None:
        trace.record(step_id, value, layer, batched)
```

```python
"""The building blocks of the Transformer. Each marked line is one traceable step."""
from __future__ import annotations

import math

import numpy as np

from .autograd import Tensor, masked_fill, repeat_kv, rope, silu, softmax
from .trace import Tracer, record


class Module:
    def named_parameters(self, prefix: str = "") -> dict[str, Tensor]:
        params: dict[str, Tensor] = {}
        for name, value in vars(self).items():
            if isinstance(value, Tensor) and value.requires_grad:
                params[prefix + name] = value
            elif isinstance(value, Module):
                params.update(value.named_parameters(f"{prefix}{name}."))
            elif isinstance(value, list):
                for i, item in enumerate(value):
                    if isinstance(item, Module):
                        params.update(item.named_parameters(f"{prefix}{name}.{i}."))
        return params

    def parameters(self) -> list[Tensor]:
        return list(self.named_parameters().values())


def init_param(rng: np.random.Generator, shape: tuple[int, ...], std: float, dtype) -> Tensor:
    return Tensor(rng.normal(0.0, std, shape).astype(dtype), requires_grad=True)


class RMSNorm(Module):
    def __init__(self, d: int, eps: float, dtype):
        self.eps = eps
        self.gamma = Tensor(np.ones(d, dtype=dtype), requires_grad=True)

    def __call__(self, x: Tensor) -> Tensor:
        mean_square = (x * x).mean(axis=-1, keepdims=True)
        return x / (mean_square + self.eps) ** 0.5 * self.gamma


def rope_angles(d_k: int, max_len: int, base: float) -> tuple[np.ndarray, np.ndarray]:
    """cos and sin of m * theta_i, with theta_i = base^(-2i/d_k). Shape (max_len, d_k/2)."""
    i = np.arange(d_k // 2)
    theta = base ** (-2.0 * i / d_k)
    angles = np.outer(np.arange(max_len), theta)
    return np.cos(angles), np.sin(angles)


def causal_mask(n_queries: int, n_keys: int, start: int) -> np.ndarray:
    """True where a query may NOT look: key position j is after query position start + i."""
    query_pos = start + np.arange(n_queries)[:, None]
    key_pos = np.arange(n_keys)[None, :]
    return key_pos > query_pos


class KVCache:
    """Keys and values already computed for earlier positions, per layer."""

    def __init__(self):
        self.k: np.ndarray | None = None
        self.v: np.ndarray | None = None


class Attention(Module):
    def __init__(self, d: int, n_heads: int, n_kv_heads: int, max_len: int, rope_base: float, rng, std, dtype):
        self.n_heads, self.n_kv_heads = n_heads, n_kv_heads
        self.d_k = d // n_heads
        self.W_Q = init_param(rng, (d, n_heads * self.d_k), std, dtype)
        self.W_K = init_param(rng, (d, n_kv_heads * self.d_k), std, dtype)
        self.W_V = init_param(rng, (d, n_kv_heads * self.d_k), std, dtype)
        self.W_O = init_param(rng, (n_heads * self.d_k, d), std, dtype)
        cos, sin = rope_angles(self.d_k, max_len, rope_base)
        self.cos, self.sin = cos.astype(dtype), sin.astype(dtype)

    def _split_heads(self, x: Tensor, heads: int) -> Tensor:
        b, n, _ = x.shape
        return x.reshape(b, n, heads, self.d_k).transpose(0, 2, 1, 3)  # (B, heads, n, d_k)

    def __call__(self, x: Tensor, layer: int, trace: Tracer | None = None,
                 cache: KVCache | None = None, start: int = 0) -> Tensor:
        b, n, _ = x.shape
        q = self._split_heads(x @ self.W_Q, self.n_heads)  # @step attn.q
        k = self._split_heads(x @ self.W_K, self.n_kv_heads)  # @step attn.k
        v = self._split_heads(x @ self.W_V, self.n_kv_heads)  # @step attn.v
        record(trace, "attn.q", q, layer)
        record(trace, "attn.k", k, layer)
        record(trace, "attn.v", v, layer)

        cos, sin = self.cos[start:start + n], self.sin[start:start + n]
        q = rope(q, cos, sin)  # @step attn.rope_q
        k = rope(k, cos, sin)  # @step attn.rope_k
        record(trace, "attn.rope_q", q, layer)
        record(trace, "attn.rope_k", k, layer)

        if cache is not None:
            # Inference only: glue the new keys and values after the cached ones.
            if cache.k is not None:
                k = Tensor(np.concatenate([cache.k, k.data], axis=2))
                v = Tensor(np.concatenate([cache.v, v.data], axis=2))
            cache.k, cache.v = k.data, v.data

        repeats = self.n_heads // self.n_kv_heads
        k_all, v_all = repeat_kv(k, repeats), repeat_kv(v, repeats)

        scores = (q @ k_all.transpose(0, 1, 3, 2)) / math.sqrt(self.d_k)  # @step attn.scores
        record(trace, "attn.scores", scores, layer)
        mask = causal_mask(n, k_all.shape[2], start)
        scores = masked_fill(scores, mask, -np.inf)  # @step attn.mask
        record(trace, "attn.mask", scores, layer)
        weights = softmax(scores, axis=-1)  # @step attn.softmax
        record(trace, "attn.softmax", weights, layer)
        mixed = weights @ v_all  # @step attn.mix
        record(trace, "attn.mix", mixed, layer)

        concat = mixed.transpose(0, 2, 1, 3).reshape(b, n, self.n_heads * self.d_k)
        out = concat @ self.W_O  # @step attn.out
        record(trace, "attn.out", out, layer)
        return out


class SwiGLU(Module):
    def __init__(self, d: int, rng, std, dtype):
        hidden = int(8 * d / 3)  # three matrices with ~8d^2 parameters in total
        self.W1 = init_param(rng, (d, hidden), std, dtype)
        self.W3 = init_param(rng, (d, hidden), std, dtype)
        self.W2 = init_param(rng, (hidden, d), std, dtype)

    def __call__(self, x: Tensor, layer: int, trace: Tracer | None = None) -> Tensor:
        gate = x @ self.W1  # @step ffn.gate
        up = x @ self.W3  # @step ffn.up
        act = silu(gate) * up  # @step ffn.act
        down = act @ self.W2  # @step ffn.down
        record(trace, "ffn.gate", gate, layer)
        record(trace, "ffn.up", up, layer)
        record(trace, "ffn.act", act, layer)
        record(trace, "ffn.down", down, layer)
        return down


class Block(Module):
    def __init__(self, cfg, rng, dtype):
        self.norm1 = RMSNorm(cfg.d_model, cfg.eps, dtype)
        self.attn = Attention(cfg.d_model, cfg.n_heads, cfg.n_kv_heads, cfg.max_len, cfg.rope_base,
                              rng, cfg.init_std, dtype)
        self.norm2 = RMSNorm(cfg.d_model, cfg.eps, dtype)
        self.ffn = SwiGLU(cfg.d_model, rng, cfg.init_std, dtype)

    def __call__(self, x: Tensor, layer: int, trace: Tracer | None = None,
                 cache: KVCache | None = None, start: int = 0) -> Tensor:
        h = self.norm1(x)  # @step norm.attn
        record(trace, "norm.attn", h, layer)
        x = x + self.attn(h, layer, trace, cache, start)  # @step resid.attn
        record(trace, "resid.attn", x, layer)
        h = self.norm2(x)  # @step norm.ffn
        record(trace, "norm.ffn", h, layer)
        x = x + self.ffn(h, layer, trace)  # @step resid.ffn
        record(trace, "resid.ffn", x, layer)
        return x
```

```python
"""The same model written with PyTorch, used only to check glassllm in tests."""
import numpy as np
import torch
import torch.nn.functional as F

from glassllm.nn import rope_angles


def rmsnorm(x, gamma, eps):
    return gamma * x / torch.sqrt((x * x).mean(-1, keepdim=True) + eps)


def rope(x, cos, sin):
    x_even, x_odd = x[..., 0::2], x[..., 1::2]
    return torch.stack([x_even * cos - x_odd * sin, x_even * sin + x_odd * cos], dim=-1).flatten(-2)


def to_torch(model):
    """Copy glassllm parameters into float64 torch tensors that track gradients."""
    return {name: torch.tensor(p.data, dtype=torch.float64, requires_grad=True)
            for name, p in model.named_parameters().items()}


def forward(params, ids, cfg):
    ids = torch.as_tensor(np.atleast_2d(ids))
    b, n = ids.shape
    h_q, h_kv = cfg.n_heads, cfg.n_kv_heads
    d_k = cfg.d_model // h_q
    cos, sin = (torch.tensor(a[:n]) for a in rope_angles(d_k, cfg.max_len, cfg.rope_base))
    x = params["E"][ids]
    for i in range(cfg.n_layers):
        p = lambda name: params[f"blocks.{i}.{name}"]  # noqa: E731
        h = rmsnorm(x, p("norm1.gamma"), cfg.eps)
        q = (h @ p("attn.W_Q")).view(b, n, h_q, d_k).transpose(1, 2)
        k = (h @ p("attn.W_K")).view(b, n, h_kv, d_k).transpose(1, 2)
        v = (h @ p("attn.W_V")).view(b, n, h_kv, d_k).transpose(1, 2)
        q, k = rope(q, cos, sin), rope(k, cos, sin)
        k = k.repeat_interleave(h_q // h_kv, dim=1)
        v = v.repeat_interleave(h_q // h_kv, dim=1)
        att = F.scaled_dot_product_attention(q, k, v, is_causal=True)
        x = x + att.transpose(1, 2).reshape(b, n, h_q * d_k) @ p("attn.W_O")
        h = rmsnorm(x, p("norm2.gamma"), cfg.eps)
        x = x + (F.silu(h @ p("ffn.W1")) * (h @ p("ffn.W3"))) @ p("ffn.W2")
    h = rmsnorm(x, params["norm_f.gamma"], cfg.eps)
    return h @ params["E"].T
```

```python
"""GlassLM: a decoder-only Transformer (RMSNorm, RoPE, GQA, SwiGLU, tied embeddings)."""
from __future__ import annotations

import json
from dataclasses import asdict
from pathlib import Path

import numpy as np

from .autograd import Tensor, cross_entropy, cross_entropy_grad, embedding, transpose
from .config import Config
from .nn import Block, KVCache, Module, RMSNorm, init_param
from .trace import Tracer, record


class GlassLM(Module):
    def __init__(self, cfg: Config, seed: int = 0, dtype=np.float32):
        self.cfg = cfg
        rng = np.random.default_rng(seed)
        self.E = init_param(rng, (cfg.vocab_size, cfg.d_model), cfg.init_std, dtype)
        self.blocks = [Block(cfg, rng, dtype) for _ in range(cfg.n_layers)]
        self.norm_f = RMSNorm(cfg.d_model, cfg.eps, dtype)

    def new_cache(self) -> list[KVCache]:
        return [KVCache() for _ in self.blocks]

    def forward(self, ids, trace: Tracer | None = None,
                cache: list[KVCache] | None = None, start: int = 0) -> Tensor:
        """ids: (n,) or (B, n) token ids. Returns logits (B, n, vocab_size)."""
        ids = np.asarray(ids, dtype=np.int64)
        if ids.ndim == 1:
            ids = ids[None, :]
        b, n = ids.shape
        if start + n > self.cfg.max_len:
            raise ValueError(f"{start + n} positions do not fit in max_len={self.cfg.max_len}")
        if trace is not None and b != 1:
            raise ValueError("the tracer follows a single sequence; pass batch size 1")

        x = embedding(self.E, ids)  # @step embed
        record(trace, "embed", x)
        for i, block in enumerate(self.blocks):
            x = block(x, i, trace, None if cache is None else cache[i], start)
        h = self.norm_f(x)  # @step norm.final
        record(trace, "norm.final", h)
        logits = h @ transpose(self.E, (1, 0))  # @step logits
        record(trace, "logits", logits)
        return logits

    __call__ = forward

    def loss(self, ids, targets, trace: Tracer | None = None) -> Tensor:
        """Mean cross-entropy of predicting targets[t] from ids[..t]."""
        logits = self.forward(ids, trace)
        b, n, v = logits.shape
        flat_targets = np.asarray(targets).reshape(b * n)
        loss = cross_entropy(logits.reshape(b * n, v), flat_targets)  # @step loss
        record(trace, "loss", loss, batched=False)
        if trace is not None:
            grad = cross_entropy_grad(logits.data.reshape(b * n, v), flat_targets)
            record(trace, "loss.grad", grad.reshape(b, n, v))
        return loss

    def save(self, path: str | Path, fp16: bool = True) -> None:
        arrays = {name: p.data.astype(np.float16 if fp16 else p.data.dtype)
                  for name, p in self.named_parameters().items()}
        np.savez_compressed(path, __config__=np.array(json.dumps(asdict(self.cfg))), **arrays)

    @classmethod
    def load(cls, path: str | Path, dtype=np.float32) -> "GlassLM":
        with np.load(path) as data:
            cfg = Config(**json.loads(str(data["__config__"])))
            model = cls(cfg, dtype=dtype)
            for name, p in model.named_parameters().items():
                p.data = data[name].astype(dtype)
        return model
```

```python
import json
from dataclasses import asdict, dataclass
from pathlib import Path

            raise ValueError(f"{start + n} positions do not fit in max_len={self.cfg.max_len}")
        if trace is not None and b != 1:
            raise ValueError("the tracer follows a single sequence; pass batch size 1")

        x = embedding(self.E, ids)  # @step embed
        record(trace, "embed", x)
        for i, block in enumerate(self.blocks):
            x = block(x, i, trace, None if cache is None else cache[i], start)
        h = self.norm_f(x)  # @step norm.final
        record(trace, "norm.final", h)
        logits = h @ transpose(self.E, (1, 0))  # @step logits
        record(trace, "logits", logits)
        return logits

    __call__ = forward

    def loss(self, ids, targets, trace: Tracer | None = None) -> Tensor:
        """Mean cross-entropy of predicting targets[t] from ids[..t]."""
        logits = self.forward(ids, trace)
        b, n, v = logits.shape
        flat_targets = np.asarray(targets).reshape(b * n)
        loss = cross_entropy(logits.reshape(b * n, v), flat_targets)  # @step loss
        record(trace, "loss", loss, batched=False)
        if trace is not None:
            grad = cross_entropy_grad(logits.data.reshape(b * n, v), flat_targets)
            record(trace, "loss.grad", grad.reshape(b, n, v))
        return loss

    def generate(self, ids, n_new: int, temperature: float = 1.0, greedy: bool = False,
                 seed: int | None = None, trace: bool = False) -> Iterator[GenStep]:
        """Yield one new token at a time, using a KV cache."""
        if temperature <= 0:
            raise ValueError("temperature must be > 0; use greedy=True to always take the most likely token")
        ids = [int(i) for i in ids]
        if not ids:
            raise ValueError("the prompt is empty; give at least one token")
        max_len = self.cfg.max_len
        ids = ids[-max_len:]
        rng = np.random.default_rng(seed)
        with no_grad():
            cache, start, pending = self.new_cache(), 0, ids
            for _ in range(n_new):
                if start + len(pending) > max_len:
                    # Context is full: keep the most recent half and rebuild the cache.
                    cache, start, pending = self.new_cache(), 0, ids[-(max_len // 2):]
                tracer = Tracer() if trace else None
                logits = self.forward(np.array(pending), tracer, cache, start)
                start += len(pending)
                probs = next_token_probs(logits.data[0, -1].astype(np.float64), temperature)
                token = int(np.argmax(probs)) if greedy else int(rng.choice(len(probs), p=probs))  # @step sample
                record(tracer, "probs", probs, batched=False)
                record(tracer, "sample", np.array(token), batched=False)
                ids.append(token)
                pending = [token]
                yield GenStep(token, probs, tracer)

    def save(self, path: str | Path, fp16: bool = True) -> None:
        arrays = {name: p.data.astype(np.float16 if fp16 else p.data.dtype)
                  for name, p in self.named_parameters().items()}
        np.savez_compressed(path, __config__=np.array(json.dumps(asdict(self.cfg))), **arrays)

    @classmethod
    def load(cls, path: str | Path, dtype=np.float32) -> "GlassLM":
        with np.load(path) as data:
            cfg = Config(**json.loads(str(data["__config__"])))
            model = cls(cfg, dtype=dtype)
            for name, p in model.named_parameters().items():
                p.data = data[name].astype(dtype)
        return model
```

```python
"""AdamW, gradient clipping and the learning rate schedule."""
from __future__ import annotations

import math

import numpy as np

from .autograd import Tensor


class AdamW:
    def __init__(self, params: list[Tensor], lr: float = 3e-3, betas=(0.9, 0.95),
                 eps: float = 1e-8, weight_decay: float = 0.1):
        self.params = list(params)
        self.lr, self.betas, self.eps, self.weight_decay = lr, betas, eps, weight_decay
        self.m = [np.zeros_like(p.data) for p in self.params]
        self.v = [np.zeros_like(p.data) for p in self.params]
        self.t = 0

    def zero_grad(self) -> None:
        for p in self.params:
            p.grad = None

    def step(self, lr: float | None = None) -> None:
        lr = self.lr if lr is None else lr
        beta1, beta2 = self.betas
        self.t += 1
        for p, m, v in zip(self.params, self.m, self.v):
            if p.grad is None:
                continue
            g = p.grad
            m *= beta1
            m += (1 - beta1) * g
            v *= beta2
            v += (1 - beta2) * g * g
            m_hat = m / (1 - beta1**self.t)
            v_hat = v / (1 - beta2**self.t)
            if p.data.ndim >= 2:
                p.data -= lr * self.weight_decay * p.data  # decoupled decay, matrices only
            p.data -= (lr * m_hat / (np.sqrt(v_hat) + self.eps)).astype(p.data.dtype)


def clip_grad_norm(params: list[Tensor], max_norm: float) -> float:
    """Scale all gradients so their global norm is at most max_norm. Returns the norm before clipping."""
    grads = [p.grad for p in params if p.grad is not None]
    total = math.sqrt(sum(float((g * g).sum()) for g in grads))
    if total > max_norm:
        scale = max_norm / (total + 1e-6)
        for p in params:
            if p.grad is not None:
                p.grad = p.grad * scale
    return total


def lr_schedule(step: int, total_steps: int, base_lr: float, warmup: int, min_ratio: float = 0.1) -> float:
    """Linear warmup, then cosine decay down to min_ratio * base_lr."""
    if step < warmup:
        return base_lr * (step + 1) / warmup
    progress = (step - warmup) / max(1, total_steps - warmup)
    cosine = 0.5 * (1 + math.cos(math.pi * min(1.0, progress)))
    return base_lr * (min_ratio + (1 - min_ratio) * cosine)
```

`src/glassllm/train.py`:

```python
"""Training loop: random windows of the corpus, next-token targets, AdamW."""
from __future__ import annotations

import time
from typing import Callable

import numpy as np

from .autograd import no_grad
from .model import GlassLM
from .optim import AdamW, clip_grad_norm, lr_schedule


def get_batch(data: np.ndarray, batch_size: int, seq_len: int, rng: np.random.Generator):
    """x is a window of the corpus; y is the same window shifted one token to the right."""
    if len(data) <= seq_len:
        raise ValueError(f"need more than seq_len={seq_len} tokens, got {len(data)}")
    starts = rng.integers(0, len(data) - seq_len, size=batch_size)
    x = np.stack([data[s:s + seq_len] for s in starts])
    y = np.stack([data[s + 1:s + seq_len + 1] for s in starts])
    return x, y


def evaluate(model: GlassLM, data: np.ndarray, batch_size: int, seq_len: int,
             n_batches: int = 10, seed: int = 1234) -> float:
    rng = np.random.default_rng(seed)
    losses = []
    with no_grad():
        for _ in range(n_batches):
            x, y = get_batch(data, batch_size, seq_len, rng)
            losses.append(float(model.loss(x, y).data))
    return float(np.mean(losses))


def train(model: GlassLM, data: np.ndarray, steps: int, batch_size: int = 16, seq_len: int = 128,
          lr: float = 3e-3, warmup: int = 100, seed: int = 0, val_data: np.ndarray | None = None,
          eval_every: int = 100, keep_best: bool = False,
          log: Callable[[str], None] | None = print) -> list[dict]:
    """Train in place. With keep_best and val_data, the model ends with the weights of the best validation loss."""
    rng = np.random.default_rng(seed)
    params = model.parameters()
    opt = AdamW(params, lr=lr)
    history = []
    best_val, best_weights = float("inf"), None
    t0 = time.time()
    for step in range(steps):
        x, y = get_batch(data, batch_size, seq_len, rng)
        loss = model.loss(x, y)
        opt.zero_grad()
        loss.backward()
        clip_grad_norm(params, 1.0)
        step_lr = lr_schedule(step, steps, lr, warmup)
        opt.step(step_lr)
        entry = {"step": step, "loss": float(loss.data), "lr": step_lr}
        last = step == steps - 1
        if val_data is not None and (step % eval_every == 0 or last):
            entry["val_loss"] = evaluate(model, val_data, batch_size, seq_len)
            if keep_best and entry["val_loss"] < best_val:
                # A small corpus is memorized fast: past this point training loss falls but validation rises.
                best_val, best_weights = entry["val_loss"], [p.data.copy() for p in params]
        history.append(entry)
        if log is not None and (step % eval_every == 0 or last):
            val = f"  val {entry['val_loss']:.3f}" if "val_loss" in entry else ""
            log(f"step {step:5d}  loss {entry['loss']:.3f}{val}  lr {step_lr:.2e}  {time.time() - t0:.0f}s")
    if best_weights is not None:
        for p, w in zip(params, best_weights):
            p.data = w
        if log is not None:
            log(f"restored the weights with the best validation loss ({best_val:.3f})")
    return history
```

```python
# Project Gutenberg ids of public domain books, one list per language.
BOOKS = {

        55752: "Dom Casmurro",
        54829: "Memorias Posthumas de Braz Cubas",
        55682: "Quincas Borba",

        620: "Sylvie and Bruno",
    },

START = re.compile(r"\*\*\* ?START OF (THE|THIS) PROJECT GUTENBERG EBOOK.*?\*\*\*", re.S)
END = re.compile(r"\*\*\* ?END OF (THE|THIS) PROJECT GUTENBERG EBOOK")


def clean(raw: str) -> str:
    text = raw.replace("\r\n", "\n")
    text = text[START.search(text).end():END.search(text).start()]
    text = unicodedata.normalize("NFC", text)
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip() + "\n"


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", default="data")
    args = parser.parse_args()
    out = Path(args.out)
    out.mkdir(exist_ok=True)
    for lang, books in BOOKS.items():
        train_parts, val_parts = [], []
        for book_id, title in books.items():
            text = clean(urllib.request.urlopen(URL.format(id=book_id)).read().decode("utf-8-sig"))
            cut = text.index("\n", int(len(text) * 0.95)) + 1  # last 5% of each book, split on a line break

            val_parts.append(text[cut:])
```

```python
    val_loss = evaluate(GlassLM.load(path), val_ids, args.batch_size, cfg.max_len)
    best_fp32 = min(h["val_loss"] for h in history if "val_loss" in h)
    meta = {"steps": args.steps, "val_loss_fp32": best_fp32, "val_loss": val_loss,
            "train_tokens": int(len(train_ids)), "val_tokens": int(len(val_ids))}
    (out / f"{args.lang}.meta.json").write_text(json.dumps(meta, indent=2) + "\n", encoding="utf-8")
    print(f"saved {path}: val loss {val_loss:.3f} (fp16)")


if __name__ == "__main__":
    main()
```

`src/glassllm/cli.py`:

```python
"""Command line: tokenize, generate, trace and train from the terminal."""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

import numpy as np

from . import load_pretrained
from .model import Config, GlassLM
from .steps import HEAD_STEPS, LAYER_STEPS, STEPS
from .tokenizer import Tokenizer
from .trace import Tracer
from .train import train

MAX_COLS = 8


def _label(token: str) -> str:
    return token.replace("\n", "\\n")


def format_matrix(values: np.ndarray, row_labels: list[str], col_labels: list[str] | None = None) -> str:
    """Rows labeled by token; at most MAX_COLS columns, the rest summarized as '…'."""
    cols = values.shape[1]
    shown = min(cols, MAX_COLS)
    if col_labels is None:
        col_labels = [str(j) for j in range(cols)]
    width = max(8, *(len(c) for c in col_labels[:shown]))
    left = max(len(r) for r in row_labels)
    header = " " * left + " " + " ".join(c[:width].rjust(width) for c in col_labels[:shown])
    lines = [header + ("  …" if cols > shown else "")]
    for label, row in zip(row_labels, values):
        cells = " ".join(f"{v:{width}.3f}" if np.isfinite(v) else "-inf".rjust(width) for v in row[:shown])
        lines.append(label.rjust(left) + " " + cells + ("  …" if cols > shown else ""))
    return "\n".join(lines)


def cmd_tokenize(args, model, tok) -> None:
    ids = tok.encode(args.text)
    for token, i in zip(tok.tokens(ids), ids):
        print(f"{i:5d}  {_label(token)}")
    if args.merges:
        print()
        for entry in tok.encode_verbose(args.text):
            print(" → ".join(" ".join(_label(s) for s in state) for state in entry["steps"]))


def cmd_generate(args, model, tok) -> None:
    ids = tok.encode(args.text)
    print(args.text, end="", flush=True)
    for step in model.generate(ids, args.n, temperature=args.temperature, greedy=args.greedy, seed=args.seed):
        print(tok.decode([step.token]), end="", flush=True)
    print()


def cmd_trace(args, model, tok) -> None:
    ids = tok.encode(args.text)
    labels = [_label(t) for t in tok.tokens(ids)]
    step_id = args.step
    tracer = Tracer()
    if step_id in ("probs", "sample"):
        tracer = next(model.generate(ids, 1, greedy=True, trace=True)).trace
    elif step_id in ("loss", "loss.grad"):
        if len(ids) < 2:
            raise SystemExit("loss needs at least two tokens")
        model.loss(np.array(ids[:-1]), np.array(ids[1:]), trace=tracer)
        labels = labels[:-1]
    else:
        model.forward(ids, trace=tracer)

    layer = args.layer if step_id in LAYER_STEPS else None
    head = args.head if step_id in HEAD_STEPS else None
    rec = tracer.get(step_id, layer=layer, head=head)
    where = "".join([f"  layer {layer}" if layer is not None else "", f"  head {head}" if head is not None else ""])
    print(f"{step_id}{where}")
    print(f"  {rec.equation}")
    print(f"  shape {rec.shape}\n")

    value = rec.value
    if step_id == "probs":
        top = np.argsort(value)[::-1][:10]
        for i in top:
            print(f"{_label(tok.vocab[i]):>14}  {value[i]:.3f}")
    elif step_id == "sample":
        print(f"next token: {_label(tok.vocab[int(value)])!r} (id {int(value)})")
    elif value.ndim == 0:
        print(f"{float(value):.4f}")
    elif value.ndim == 2 and value.shape[1] == len(labels) and step_id.startswith("attn."):
        print(format_matrix(value, labels, labels))
    elif value.ndim == 2:
        print(format_matrix(value, labels))


def cmd_steps(args, model, tok) -> None:
    for step_id, info in STEPS.items():
        print(f"{step_id:14} {info.plain}")


def cmd_train(args, model, tok) -> None:
    text = Path(args.data).read_text(encoding="utf-8")
    data = np.array(tok.encode(text))
    cfg = Config(vocab_size=tok.vocab_size, d_model=args.d_model, n_layers=args.layers,
                 n_heads=4, n_kv_heads=2, max_len=args.seq_len)
    fresh = GlassLM(cfg, seed=args.seed)
    train(fresh, data, steps=args.steps, batch_size=args.batch_size, seq_len=args.seq_len,
          lr=args.lr, warmup=min(100, args.steps // 10 + 1), seed=args.seed, eval_every=args.log_every)
    if args.out:
        fresh.save(args.out)
        print(f"saved {args.out}")


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="glassllm", description="A small Transformer built to be looked at.")
    parser.add_argument("--lang", default="en", choices=["pt", "en"], help="which pretrained model to use")
    parser.add_argument("--model-dir", default=None, help=argparse.SUPPRESS)
    sub = parser.add_subparsers(dest="command", required=True)

    p = sub.add_parser("tokenize", help="split text into BPE tokens")
    p.add_argument("text")
    p.add_argument("--merges", action="store_true", help="show every merge, from characters to tokens")
    p.set_defaults(func=cmd_tokenize)

    p = sub.add_parser("generate", help="continue a text, one token at a time")
    p.add_argument("text")
    p.add_argument("-n", type=int, default=40, help="number of new tokens")
    p.add_argument("--temperature", type=float, default=0.8)
    p.add_argument("--greedy", action="store_true", help="always take the most likely token")
    p.add_argument("--seed", type=int, default=None)
    p.set_defaults(func=cmd_generate)

    p = sub.add_parser("trace", help="print the equation and the values of one step")
    p.add_argument("text")
    p.add_argument("--step", default="attn.softmax", choices=list(STEPS))
    p.add_argument("--layer", type=int, default=0)
    p.add_argument("--head", type=int, default=0)
    p.set_defaults(func=cmd_trace)

    p = sub.add_parser("steps", help="list every traceable step and its equation")
    p.set_defaults(func=cmd_steps)

    p = sub.add_parser("train", help="train a small model from scratch on a text file")
    p.add_argument("data", help="path to a UTF-8 text file")
    p.add_argument("--steps", type=int, default=200)
    p.add_argument("--batch-size", type=int, default=8)
    p.add_argument("--seq-len", type=int, default=64)
    p.add_argument("--d-model", type=int, default=64)
    p.add_argument("--layers", type=int, default=2)
    p.add_argument("--lr", type=float, default=3e-3)
    p.add_argument("--seed", type=int, default=0)
    p.add_argument("--log-every", type=int, default=20)
    p.add_argument("--out", default=None, help="where to save the weights (.npz)")
    p.set_defaults(func=cmd_train)
    return parser


def main(argv: list[str] | None = None) -> None:
    args = build_parser().parse_args(argv)
    if args.command == "steps":
        args.func(args, None, None)
        return
    model, tok = load_pretrained(args.lang, args.model_dir)
    try:
        args.func(args, model, tok)
    except ValueError as err:
        raise SystemExit(f"error: {err}") from err


if __name__ == "__main__":
    main(sys.argv[1:])
```
